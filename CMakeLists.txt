cmake_minimum_required(VERSION 3.20)
project(PiCDPlayer VERSION 0.1.0 LANGUAGES CXX)
add_executable(cdplayerd src/main.cpp src/cec_device.cpp src/optical_drive.cpp src/cd_device.cpp src/disc_toc.cpp)
target_include_directories(cdplayerd PRIVATE include)
target_compile_features(cdplayerd PRIVATE cxx_std_20)
set_target_properties(cdplayerd PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(cdplayerd PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
    add_executable(optical_drive_test tests/optical_drive_test.cpp src/optical_drive.cpp)
    target_include_directories(optical_drive_test PRIVATE include)
    target_compile_features(optical_drive_test PRIVATE cxx_std_20)
    set_target_properties(optical_drive_test PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(optical_drive_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME optical_drive COMMAND optical_drive_test)
    add_executable(disc_toc_test tests/disc_toc_test.cpp src/disc_toc.cpp)
    target_include_directories(disc_toc_test PRIVATE include)
    target_compile_features(disc_toc_test PRIVATE cxx_std_20)
    set_target_properties(disc_toc_test PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(disc_toc_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME disc_toc COMMAND disc_toc_test)
endif()
