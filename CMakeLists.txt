cmake_minimum_required(VERSION 3.20)
project(PiCDPlayer VERSION 0.1.0 LANGUAGES CXX)
option(ENABLE_PARANOIA "Enable libcdio-paranoia reader" OFF)
option(ENABLE_METADATA "Enable MusicBrainz metadata support" OFF)
option(ENABLE_API "Enable the local control API" OFF)
option(INSTALL_SYSTEMD_UNIT "Install the picdplayer systemd service" OFF)
set(PICDPLAYER_SERVICE_USER "picdplayer" CACHE STRING "User for the systemd service")
include(GNUInstallDirs)
set(PICDPLAYER_SYSTEMD_UNIT_DIR "lib/systemd/system" CACHE STRING
    "Systemd unit install directory (relative to install prefix or absolute)")
if(ENABLE_PARANOIA OR ENABLE_METADATA)
    find_package(PkgConfig REQUIRED)
endif()
if(ENABLE_PARANOIA)
    pkg_check_modules(PARANOIA QUIET IMPORTED_TARGET libcdio_paranoia)
    if(NOT PARANOIA_FOUND)
        message(FATAL_ERROR "ENABLE_PARANOIA=ON requires libcdio_paranoia (Debian: libcdio-paranoia-dev)")
    endif()
endif()
if(ENABLE_METADATA)
    pkg_check_modules(DISCID QUIET IMPORTED_TARGET libdiscid)
    if(NOT DISCID_FOUND)
        message(FATAL_ERROR "ENABLE_METADATA=ON requires libdiscid (Debian: libdiscid-dev)")
    endif()
    find_package(CURL QUIET)
    if(NOT CURL_FOUND)
        message(FATAL_ERROR "ENABLE_METADATA=ON requires libcurl (Debian: libcurl4-openssl-dev)")
    endif()
endif()
if(ENABLE_METADATA OR ENABLE_API)
    find_package(nlohmann_json CONFIG QUIET)
    if(NOT nlohmann_json_FOUND)
        message(FATAL_ERROR "ENABLE_METADATA or ENABLE_API requires nlohmann_json (Debian: nlohmann-json3-dev)")
    endif()
endif()
if(ENABLE_API)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WEBSOCKETS QUIET IMPORTED_TARGET libwebsockets)
    if(NOT WEBSOCKETS_FOUND)
        message(FATAL_ERROR "ENABLE_API=ON requires libwebsockets (Debian: libwebsockets-dev)")
    endif()
endif()
find_package(ALSA REQUIRED)
find_package(Threads REQUIRED)
add_executable(cdplayerd src/main.cpp src/cec_device.cpp src/cec_input.cpp src/optical_drive.cpp src/cd_device.cpp src/disc_toc.cpp src/cdda_reader.cpp src/cdda_probe.cpp src/pcm_file.cpp src/player_controller.cpp src/pcm_worker.cpp src/alsa_output.cpp src/playback_engine.cpp src/player_session.cpp src/media_state.cpp src/media_worker.cpp src/daemon_snapshot.cpp)
target_include_directories(cdplayerd PRIVATE include)
target_link_libraries(cdplayerd PRIVATE ALSA::ALSA Threads::Threads)
if(ENABLE_PARANOIA)
    target_sources(cdplayerd PRIVATE src/paranoia_reader.cpp)
    target_compile_definitions(cdplayerd PRIVATE ENABLE_PARANOIA=1)
    target_link_libraries(cdplayerd PRIVATE PkgConfig::PARANOIA)
endif()
if(ENABLE_METADATA)
    target_sources(cdplayerd PRIVATE src/musicbrainz_disc_id.cpp src/metadata_probe.cpp
        src/metadata_parser.cpp src/http_client.cpp src/metadata_lookup.cpp src/metadata_worker.cpp
        src/metadata_session.cpp)
    target_compile_definitions(cdplayerd PRIVATE ENABLE_METADATA=1)
    target_link_libraries(cdplayerd PRIVATE PkgConfig::DISCID CURL::libcurl nlohmann_json::nlohmann_json)
endif()
if(ENABLE_API)
    target_sources(cdplayerd PRIVATE src/daemon_snapshot_json.cpp src/api_server.cpp)
    target_compile_definitions(cdplayerd PRIVATE ENABLE_API=1)
    target_link_libraries(cdplayerd PRIVATE nlohmann_json::nlohmann_json PkgConfig::WEBSOCKETS)
endif()
target_compile_features(cdplayerd PRIVATE cxx_std_20)
set_target_properties(cdplayerd PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(cdplayerd PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS cdplayerd RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
configure_file(systemd/picdplayer.service.in picdplayer.service @ONLY)
if(INSTALL_SYSTEMD_UNIT)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/picdplayer.service
            DESTINATION ${PICDPLAYER_SYSTEMD_UNIT_DIR})
endif()
