cmake_minimum_required(VERSION 3.20)
project(PiCDPlayer VERSION 0.1.0 LANGUAGES CXX)
option(ENABLE_PARANOIA "Enable libcdio-paranoia reader" OFF)
if(ENABLE_PARANOIA)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PARANOIA QUIET IMPORTED_TARGET libcdio_paranoia)
    if(NOT PARANOIA_FOUND)
        message(FATAL_ERROR "ENABLE_PARANOIA=ON requires libcdio_paranoia (Debian: libcdio-paranoia-dev)")
    endif()
endif()
add_executable(cdplayerd src/main.cpp src/cec_device.cpp src/optical_drive.cpp src/cd_device.cpp src/disc_toc.cpp src/cdda_reader.cpp src/cdda_probe.cpp src/pcm_file.cpp)
target_include_directories(cdplayerd PRIVATE include)
if(ENABLE_PARANOIA)
    target_sources(cdplayerd PRIVATE src/paranoia_reader.cpp)
    target_compile_definitions(cdplayerd PRIVATE ENABLE_PARANOIA=1)
    target_link_libraries(cdplayerd PRIVATE PkgConfig::PARANOIA)
endif()
target_compile_features(cdplayerd PRIVATE cxx_std_20)
set_target_properties(cdplayerd PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(cdplayerd PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
    add_executable(optical_drive_test tests/optical_drive_test.cpp src/optical_drive.cpp)
    target_include_directories(optical_drive_test PRIVATE include)
    target_compile_features(optical_drive_test PRIVATE cxx_std_20)
    set_target_properties(optical_drive_test PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(optical_drive_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME optical_drive COMMAND optical_drive_test)
    add_executable(disc_toc_test tests/disc_toc_test.cpp src/disc_toc.cpp)
    target_include_directories(disc_toc_test PRIVATE include)
    target_compile_features(disc_toc_test PRIVATE cxx_std_20)
    set_target_properties(disc_toc_test PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(disc_toc_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME disc_toc COMMAND disc_toc_test)
    add_executable(cdda_reader_test tests/cdda_reader_test.cpp src/cdda_reader.cpp)
    target_include_directories(cdda_reader_test PRIVATE include src)
    target_compile_features(cdda_reader_test PRIVATE cxx_std_20)
    target_compile_options(cdda_reader_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME cdda_reader COMMAND cdda_reader_test)
    add_executable(pcm_file_test tests/pcm_file_test.cpp src/pcm_file.cpp)
    target_include_directories(pcm_file_test PRIVATE include)
    target_compile_features(pcm_file_test PRIVATE cxx_std_20)
    target_compile_options(pcm_file_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME pcm_file COMMAND pcm_file_test)
    if(ENABLE_PARANOIA)
        add_executable(paranoia_reader_test tests/paranoia_reader_test.cpp src/cdda_reader.cpp src/paranoia_reader.cpp)
        target_include_directories(paranoia_reader_test PRIVATE include)
        target_compile_features(paranoia_reader_test PRIVATE cxx_std_20)
        target_compile_definitions(paranoia_reader_test PRIVATE ENABLE_PARANOIA=1)
        target_compile_options(paranoia_reader_test PRIVATE -Wall -Wextra -Wpedantic)
        target_link_libraries(paranoia_reader_test PRIVATE PkgConfig::PARANOIA)
        foreach(symbol cdio_cddap_identify cdio_cddap_open cdio_cddap_close cdio_paranoia_init cdio_paranoia_free cdio_paranoia_modeset cdio_paranoia_seek cdio_paranoia_read_limited)
            target_link_options(paranoia_reader_test PRIVATE "LINKER:--wrap=${symbol}")
        endforeach()
        add_test(NAME paranoia_reader COMMAND paranoia_reader_test)
    endif()
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cdda_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cdda_cli.py $<TARGET_FILE:cdplayerd> ${ENABLE_PARANOIA})
    endif()
endif()
